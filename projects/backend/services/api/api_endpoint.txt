# ==================================================
# PRODUCT MANAGEMENT ENDPOINTS
# ==================================================

# 1. List All Products (Simple)
# GET /products/list-all-product
# Purpose: Get all products with minimal info (ID + name only)
# Response: [{"product_id": 1, "name": "Product Name"}]

curl -X GET "http://localhost:8136/products/list-all-product"

# 2. Search Products
# GET /products?name={search_term}&limit={count}
# Purpose: Search products by name with optional limit
# Parameters:
#   - name: Search term (optional)
#   - limit: Result limit (default: 50)

curl -X GET "http://localhost:8136/products?name=milk&limit=20"

# 3. Get Product Detail
# GET /products/{product_id}
# Purpose: Get complete product information including metadata
# Response: Full product details with type, area, manufacturer, etc.

curl -X GET "http://localhost:8136/products/1"

# 4. Get Products by Area
# GET /products/by-area/{area}?limit={count}
# Purpose: Filter products by store area/section
# Parameters:
#   - area: Store area name
#   - limit: Result limit (default: 50)

curl -X GET "http://localhost:8136/products/by-area/dairy?limit=30"

# 5. Get Products by Type
# GET /products/by-type/{product_type}?limit={count}
# Purpose: Filter products by category/type
# Parameters:
#   - product_type: Product category
#   - limit: Result limit (default: 50)

curl -X GET "http://localhost:8136/products/by-type/beverage?limit=25"

# 6. Update Product
# POST /products/update-product/{product_id}
# Purpose: Update existing product details
# Content-Type: application/json
# Body: ProductUpdate schema with all product fields

curl -X POST "http://localhost:8136/products/update-product/1" \
  -H "Content-Type: application/json" \
  -d '{
    "product_name": "Updated Product Name",
    "product_type": "food",
    "area": "grocery",
    "price": 15.99,
    "quantity": 100.0,
    "weight_kg": 0.5
  }'

# 7. Delete Product
# POST /products/delete-product/{product_id}
# Purpose: Remove product from system
# Response: Success confirmation

curl -X POST "http://localhost:8136/products/delete-product/1"

# ==================================================
# CART INVENTORY MANAGEMENT
# ==================================================

# 1. Update Cart Items (Replace Mode)
# POST /cart_inventory/{cart_id}/items
# Purpose: Replace entire cart contents with new items
# Content-Type: application/json

curl -X POST "http://localhost:8136/cart_inventory/1/items" \
  -H "Content-Type: application/json" \
  -d '{
    "mode": "replace",
    "items": [
      {"product_id": 1, "quantity": 2.5},
      {"product_id": 2, "quantity": 1.0}
    ]
  }'

# 2. Update Cart Items (Merge Mode)
# POST /cart_inventory/{cart_id}/items
# Purpose: Update specific items while keeping others unchanged
# Content-Type: application/json

curl -X POST "http://localhost:8136/cart_inventory/1/items" \
  -H "Content-Type: application/json" \
  -d '{
    "mode": "merge",
    "items": [
      {"product_id": 1, "quantity": 3.0}
    ]
  }'

# 3. Update Cart Items (Delta Mode)
# POST /cart_inventory/{cart_id}/items
# Purpose: Apply incremental changes (add/subtract quantities)
# Content-Type: application/json

curl -X POST "http://localhost:8136/cart_inventory/1/items" \
  -H "Content-Type: application/json" \
  -d '{
    "mode": "delta",
    "changes": [
      {"product_id": 1, "delta": 1.0},
      {"product_id": 2, "delta": -0.5}
    ]
  }'

# 4. Get Cart Details
# GET /cart_inventory/{cart_id}
# Purpose: Retrieve cart contents with calculated totals
# Response: Cart items with product details and total amount

curl -X GET "http://localhost:8136/cart_inventory/1"

# USAGE NOTES
# ==================================================
# - API service runs on port 8136 by default
# - All endpoints support JSON request/response format
# - WebSocket connections auto-reconnect on failure
# - Cart inventory supports decimal quantities for weight-based products
# - Map coordinates use integer grid system
# - Beacon coordinates use decimal precision for accurate positioning
# - Real-time streams optimized for minimal bandwidth usage
# - CORS enabled for cross-origin browser requests
# - Health check endpoint useful for monitoring and load balancers

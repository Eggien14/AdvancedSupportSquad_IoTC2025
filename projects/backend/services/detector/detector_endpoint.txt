BASE_URL=http://localhost:8100
API_PREFIX=/api/v1

# ==================================================
# ENDPOINT SPECIFICATIONS
# ==================================================

# 1. Health Check Endpoint
# GET /healthz
# Purpose: Verify service status and availability
# Response: {"status": "healthy", "service": "detector"}

curl -X GET "http://localhost:8100/healthz"

# 2. Real-time Image Detection
# POST /api/v1/image
# Purpose: Detect objects in uploaded image using YOLOv8
# Content-Type: multipart/form-data
# Parameters:
#   - file: Image file (JPG, PNG, JPEG)
#   - confidence: Detection confidence threshold (0.0-1.0, default: 0.5)

curl -X POST "http://localhost:8100/api/v1/image" \
  -F "file=@/path/to/image.jpg" \
  -F "confidence=0.7"

# 3. Inventory Detection Analysis
# POST /api/v1/inventory
# Purpose: Analyze inventory items from image with detailed metadata
# Content-Type: multipart/form-data
# Parameters:
#   - file: Image file (JPG, PNG, JPEG)
#   - confidence: Detection confidence threshold (0.0-1.0, default: 0.5)
#   - include_metadata: Include detection metadata (true/false, default: false)

curl -X POST "http://localhost:8100/api/v1/inventory" \
  -F "file=@/path/to/inventory.jpg" \
  -F "confidence=0.6" \
  -F "include_metadata=true"
